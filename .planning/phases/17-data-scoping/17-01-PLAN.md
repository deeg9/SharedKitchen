**Create `src/hooks/useKitchenDataSync.ts`:**

This hook subscribes to `activeHouseholdId` from `useHouseholdStore` and, when it changes (and is non-null), clears all data stores then re-fetches their data. This ensures ALL tabs get fresh data on kitchen switch, even if not currently focused.

```typescript
import { useEffect, useRef } from 'react';
import { useHouseholdStore } from '../stores/householdStore';
import { usePlanStore } from '../stores/planStore';
import { useRecipeStore } from '../stores/recipeStore';
import { useGroceryStore } from '../stores/groceryStore';
import { useBudgetStore } from '../stores/budgetStore';
import { getWeekDays, toDateString } from '../lib/dates';

export function useKitchenDataSync() {
  const activeHouseholdId = useHouseholdStore((s) => s.activeHouseholdId);
  const prevIdRef = useRef<string | null>(null);

  useEffect(() => {
    // Skip initial mount (tabs handle their own initial fetch)
    // Only fire on CHANGE of activeHouseholdId
    if (prevIdRef.current === null) {
      prevIdRef.current = activeHouseholdId;
      return;
    }

    // If same id, skip
    if (prevIdRef.current === activeHouseholdId) return;

    prevIdRef.current = activeHouseholdId;

    if (!activeHouseholdId) return;

    // Clear all stores to prevent stale data flash
    usePlanStore.getState().clear();
    useRecipeStore.getState().clear();
    useGroceryStore.getState().clear();
    useBudgetStore.getState().clear();

    // Re-fetch all stores with new kitchen ID
    const weekDays = getWeekDays(0);
    const weekStart = toDateString(weekDays[0]);
    const now = new Date();
    const currentMonth = new Date(now.getFullYear(), now.getMonth(), 1);

    usePlanStore.getState().fetchMealPlan(activeHouseholdId, weekStart);
    useRecipeStore.getState().fetchRecipes(activeHouseholdId);
    useGroceryStore.getState().fetchGroceryList(activeHouseholdId, weekStart);
    useBudgetStore.getState().fetchExpenses(activeHouseholdId, currentMonth);
    useBudgetStore.getState().fetchBudgetTarget(activeHouseholdId);
  }, [activeHouseholdId]);
}
```

Key design decisions:
- Uses `useRef` to track previous ID and skip the initial mount (each tab already handles its own initial data load via its own useEffect).
- Uses `getState()` for imperative store access — no need to subscribe to all stores in React.
- Fires all fetches in parallel (no await) since they are independent.
- Clears stores BEFORE fetching to avoid showing old kitchen data during load.
