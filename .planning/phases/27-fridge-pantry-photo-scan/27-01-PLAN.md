```typescript
import { PantryItem } from '../types/database';
import { ScannedItem } from '../types/scan';

export function normalizeName(name: string): string {
  return name
    .toLowerCase()
    .trim()
    .replace(/[^a-z0-9\s]/g, '')  // strip punctuation
    .replace(/\s+/g, ' ');         // collapse whitespace
}

export function findDuplicates(
  scannedItems: ScannedItem[],
  existingItems: PantryItem[]
): ScannedItem[] {
  const existingNormalized = existingItems.map(item => ({
    id: item.id,
    name: item.name,
    normalized: normalizeName(item.name),
  }));

  return scannedItems.map(scanned => {
    const scannedNorm = normalizeName(scanned.name);
    const match = existingNormalized.find(existing =>
      existing.normalized === scannedNorm ||
      existing.normalized.includes(scannedNorm) ||
      scannedNorm.includes(existing.normalized)
    );
    if (match) {
      return { ...scanned, duplicateOfId: match.id, duplicateOfName: match.name };
    }
    return scanned;
  });
}
```

Use simple substring matching -- Levenshtein is overkill for food names. The `includes` check handles cases like "Whole Milk" matching "Milk" or "2% Milk" matching "Milk".

```typescript
bulkAddPantryItems: async (householdId, items) => {
  try {
    const { data: { user } } = await supabase.auth.getUser();

    const inserts = items.map((item, idx) => ({
      household_id: householdId,
      name: item.name,
      quantity: item.quantity ?? null,
      unit: null,
      category: item.category ?? 'Other',
      created_by: user?.id ?? null,
    }));

    // Optimistic: create temp items
    const tempItems: PantryItem[] = inserts.map((ins, idx) => ({
      id: `temp-bulk-${Date.now()}-${idx}`,
      ...ins,
      added_at: new Date().toISOString(),
      expires_at: null,
      source_grocery_item_id: null,
    }));
    set((state) => ({ items: [...tempItems, ...state.items] }));

    // Single batch insert
    const { data, error } = await supabase
      .from('pantry_items')
      .insert(inserts)
      .select();

    if (error) {
      // Revert all temp items
      const tempIds = new Set(tempItems.map(t => t.id));
      set((state) => ({
        items: state.items.filter((i) => !tempIds.has(i.id)),
        error: error.message,
      }));
      return { addedCount: 0 };
    }

    // Replace temp items with real ones
    const tempIds = new Set(tempItems.map(t => t.id));
    set((state) => ({
      items: [
        ...(data as PantryItem[]),
        ...state.items.filter((i) => !tempIds.has(i.id)),
      ],
    }));
    return { addedCount: data.length };
  } catch (error: unknown) {
    set({ error: getErrorMessage(error) });
    return { addedCount: 0 };
  }
},
```

Key design choices per CONTEXT.md decisions:
- Single batch insert to Supabase (not N individual calls)
- Optimistic update with temp IDs, reverts on error
- Returns `{ addedCount }` so caller can show "Added X items to On Hand" toast
- Follows existing addPantryItem pattern (auth user, optimistic, revert)
