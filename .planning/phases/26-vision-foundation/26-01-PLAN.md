import * as ImageManipulator from 'expo-image-manipulator';
import { SaveFormat } from 'expo-image-manipulator';

export async function compressForVision(uri: string): Promise<{ base64: string; width: number; height: number }> {
  const result = await ImageManipulator.manipulateAsync(
    uri,
    [{ resize: { width: 1024 } }],
    { format: SaveFormat.JPEG, compress: 0.6, base64: true }
  );
  return { base64: result.base64!, width: result.width, height: result.height };
}
```

export async function analyzePhoto(
  imageUri: string,
  intent: 'scan' | 'identify'
): Promise<VisionResult> {
  const { base64 } = await compressForVision(imageUri);
  const session = await supabase.auth.getSession();
  const token = session.data.session?.access_token;
  const supabaseUrl = process.env.EXPO_PUBLIC_SUPABASE_URL;
  const anonKey = process.env.EXPO_PUBLIC_SUPABASE_ANON_KEY;

  const response = await fetch(`${supabaseUrl}/functions/v1/analyze-food-photo`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${token}`,
      'apikey': anonKey ?? '',
    },
    body: JSON.stringify({ image: base64, intent }),
  });

  let data: any;
  const contentType = response.headers.get('content-type') ?? '';
  if (contentType.includes('application/json')) {
    data = await response.json();
  } else {
    throw new Error(`Server error (${response.status}). Please try again.`);
  }

  if (!response.ok) {
    if (response.status === 429 && data?.type === 'vision_limit_exceeded') {
      throw new VisionRateLimitError(
        data.error ?? 'Monthly vision limit reached',
        data.limit ?? 5,
        data.used ?? 0
      );
    }
    throw new Error(data.error ?? `Vision analysis failed (${response.status})`);
  }

  return data as VisionResult;
}
```

1. Create `src/hooks/useCameraPermission.ts`:
```typescript
import { useState, useCallback } from 'react';
import * as ImagePicker from 'expo-image-picker';
import AsyncStorage from '@react-native-async-storage/async-storage';

const PREPROMPT_KEY = 'camera_preprompt_shown';

export function useCameraPermission() {
  const [showPrePrompt, setShowPrePrompt] = useState(false);
  const [permissionDenied, setPermissionDenied] = useState(false);

  const checkAndPrompt = useCallback(async (): Promise<boolean> => {
    const { status } = await ImagePicker.getCameraPermissionsAsync();
    if (status === 'granted') {
      setPermissionDenied(false);
      return true;
    }
    if (status === 'undetermined') {
      const dismissed = await AsyncStorage.getItem(PREPROMPT_KEY);
      if (!dismissed) {
        setShowPrePrompt(true);
        return false;
      }
      const result = await ImagePicker.requestCameraPermissionsAsync();
      setPermissionDenied(!result.granted);
      return result.granted;
    }
    // denied
    setPermissionDenied(true);
    return false;
  }, []);

  const requestAfterPrePrompt = useCallback(async (): Promise<boolean> => {
    await AsyncStorage.setItem(PREPROMPT_KEY, 'true');
    setShowPrePrompt(false);
    const result = await ImagePicker.requestCameraPermissionsAsync();
    setPermissionDenied(!result.granted);
    return result.granted;
  }, []);

  const dismissPrePrompt = useCallback(() => {
    setShowPrePrompt(false);
    AsyncStorage.setItem(PREPROMPT_KEY, 'true');
  }, []);

  return { showPrePrompt, permissionDenied, checkAndPrompt, requestAfterPrePrompt, dismissPrePrompt };
}
```
