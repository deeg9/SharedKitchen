       - `recipe_ratings` table with: id (UUID PK), recipe_id (FK), user_id (FK), rating (1-5 CHECK), is_favorite (bool default false), created_at, UNIQUE(recipe_id, user_id)
       - `recipe_notes` table with: id (UUID PK), recipe_id (FK), user_id (FK), content (TEXT NOT NULL), created_at
