3. Create `app/+native-intent.ts` -- this is Expo Router's hook for intercepting native deep links. When a share intent arrives, redirect to the add-recipe screen with the shared URL as a query parameter:

   ```typescript
   import { getShareExtensionKey } from "expo-share-intent";

   export function redirectSystemPath({
     path,
     initial,
   }: {
     path: string;
     initial: string;
   }) {
     try {
       if (path.includes(`dataUrl=${getShareExtensionKey()}`)) {
         // Share intent received -- route to add recipe screen
         // The ShareIntentProvider + useShareIntentContext in add.tsx
         // will pick up the actual URL data
         return "/(app)/recipe/add?fromShare=1";
       }
       return path;
     } catch {
       return "/";
     }
   }
   ```

   The `fromShare=1` param signals add.tsx to read the share intent data on mount. The actual shared URL comes from the useShareIntentContext hook, not from the URL param.

   ```typescript
   useEffect(() => {
     if (hasShareIntent && shareIntent?.webUrl) {
       const sharedUrl = shareIntent.webUrl;
       setImportUrl(sharedUrl);
       resetShareIntent();
       // Auto-trigger import after a brief delay for UI to render
       const timer = setTimeout(() => {
         // Directly call the import logic with the shared URL
         handleImportUrl(sharedUrl);
       }, 300);
       return () => clearTimeout(timer);
     }

   ```

   const handleImportUrl = async (url: string) => {
     const urlIsVideo = isVideoUrl(url);
     setImporting(true);
     startProgress(urlIsVideo);
     try {
       const data = await parseRecipeUrl(url);
       setImportedData(data);
       if (urlIsVideo) {
         const { data: { user } } = await supabase.auth.getUser();
         if (user) {
           const usage = await getVideoImportCount(user.id);
           setVideoUsage(usage);
         }
       }
     } catch (error: unknown) {
       if (error instanceof VideoRateLimitError) {
         setShowUpgradePrompt(true);
       } else {
         Alert.alert('Import Failed', getErrorMessage(error));
       }
     } finally {
       setImporting(false);
       stopProgress();
     }
   };
   ```

   const handleImport = async () => {
     if (!importUrl.trim()) {
       Alert.alert('Error', 'Please enter a recipe URL.');
       return;

     await handleImportUrl(importUrl.trim());
   };
   ```
