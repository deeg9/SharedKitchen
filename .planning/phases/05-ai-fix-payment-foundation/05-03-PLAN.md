**Step 1: Create `supabase/functions/create-checkout-session/index.ts`:**

```typescript
// Creates a Stripe Checkout Session for the $9.99/mo premium subscription.
// Returns clientSecret + ephemeralKey + customer for PaymentSheet.

import Stripe from 'https://esm.sh/stripe@17?target=denonext'
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'

const stripe = new Stripe(Deno.env.get('STRIPE_SECRET_KEY')!, {
  apiVersion: '2024-11-20',
})

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
}

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders })
  }

  try {
    // Authenticate user via JWT
    const authHeader = req.headers.get('Authorization')
    const supabaseClient = createClient(
      Deno.env.get('SUPABASE_URL')!,
      Deno.env.get('SUPABASE_ANON_KEY')!,
      { global: { headers: { Authorization: authHeader! } } }
    )
    const { data: { user }, error: userError } = await supabaseClient.auth.getUser()
    if (!user || userError) {
      return new Response(
        JSON.stringify({ error: 'Unauthorized' }),
        { status: 401, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
      )
    }

    // Check for existing Stripe customer in subscriptions table
    const supabaseAdmin = createClient(
      Deno.env.get('SUPABASE_URL')!,
      Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
    )
    const { data: existingSub } = await supabaseAdmin
      .from('subscriptions')
      .select('stripe_customer_id')
      .eq('user_id', user.id)
      .maybeSingle()

    let customerId = existingSub?.stripe_customer_id

    // Create or retrieve Stripe customer
    if (!customerId) {
      const customer = await stripe.customers.create({
        email: user.email,
        metadata: { user_id: user.id },
      })
      customerId = customer.id

      // Store customer ID for future use
      await supabaseAdmin
        .from('subscriptions')
        .upsert({
          user_id: user.id,
          stripe_customer_id: customerId,
          status: 'free',
        }, { onConflict: 'user_id' })
    }

    // Get price ID from env or request body
    const body = await req.json().catch(() => ({}))
    const priceId = body.priceId || Deno.env.get('STRIPE_PRICE_ID')
    if (!priceId) {
      return new Response(
        JSON.stringify({ error: 'Missing price ID' }),
        { status: 400, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
      )
    }

    // Create ephemeral key for PaymentSheet
    const ephemeralKey = await stripe.ephemeralKeys.create(
      { customer: customerId },
      { apiVersion: '2024-11-20' }
    )

    // Create checkout/subscription via PaymentIntent approach
    const subscription = await stripe.subscriptions.create({
      customer: customerId,
      items: [{ price: priceId }],
      payment_behavior: 'default_incomplete',
      payment_settings: {
        save_default_payment_method: 'on_subscription',
      },
      expand: ['latest_invoice.payment_intent'],
      metadata: { user_id: user.id },
    })

    const invoice = subscription.latest_invoice as Stripe.Invoice
    const paymentIntent = invoice.payment_intent as Stripe.PaymentIntent

    return new Response(
      JSON.stringify({
        clientSecret: paymentIntent.client_secret,
        ephemeralKey: ephemeralKey.secret,
        customer: customerId,
        subscriptionId: subscription.id,
      }),
      { status: 200, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
    )
  } catch (error) {
    console.error('Error creating checkout session:', error)
    return new Response(
      JSON.stringify({ error: 'Failed to create checkout session' }),
      { status: 500, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
    )
  }
})
```

// Creates a Stripe Customer Portal session for subscription management.

import Stripe from 'https://esm.sh/stripe@17?target=denonext'
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'

const stripe = new Stripe(Deno.env.get('STRIPE_SECRET_KEY')!, {
  apiVersion: '2024-11-20',
})

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
}

Deno.serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response('ok', { headers: corsHeaders })
  }

  try {
    // Authenticate user
    const authHeader = req.headers.get('Authorization')
    const supabaseClient = createClient(
      Deno.env.get('SUPABASE_URL')!,
      Deno.env.get('SUPABASE_ANON_KEY')!,
      { global: { headers: { Authorization: authHeader! } } }
    )
    const { data: { user }, error: userError } = await supabaseClient.auth.getUser()
    if (!user || userError) {
      return new Response(
        JSON.stringify({ error: 'Unauthorized' }),
        { status: 401, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
      )
    }

    // Get customer ID from subscriptions table
    const supabaseAdmin = createClient(
      Deno.env.get('SUPABASE_URL')!,
      Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!
    )
    const { data: sub } = await supabaseAdmin
      .from('subscriptions')
      .select('stripe_customer_id')
      .eq('user_id', user.id)
      .single()

    if (!sub?.stripe_customer_id) {
      return new Response(
        JSON.stringify({ error: 'No subscription found' }),
        { status: 404, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
      )
    }

    // Create billing portal session
    const portalSession = await stripe.billingPortal.sessions.create({
      customer: sub.stripe_customer_id,
      return_url: 'sharedkitchen://subscription',
    })

    return new Response(
      JSON.stringify({ url: portalSession.url }),
      { status: 200, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
    )
  } catch (error) {
    console.error('Error creating billing portal:', error)
    return new Response(
      JSON.stringify({ error: 'Failed to open billing portal' }),
      { status: 500, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
    )
  }
})

**Step 3: Create `src/lib/stripe.ts`:**

```typescript
import { Alert } from 'react-native';
import * as Linking from 'expo-linking';
import { initPaymentSheet, presentPaymentSheet } from '@stripe/stripe-react-native';
import { supabase } from './supabase';
import { useSubscriptionStore } from '../stores/subscriptionStore';

const STRIPE_PRICE_ID = process.env.EXPO_PUBLIC_STRIPE_PRICE_ID ?? '';

/**
 * Opens Stripe PaymentSheet for premium subscription checkout.
 * Returns true if payment succeeded, false if canceled/failed.
 */
export async function openCheckout(): Promise<boolean> {
  // 1. Create checkout session via Edge Function
  const { data, error } = await supabase.functions.invoke('create-checkout-session', {
    body: { priceId: STRIPE_PRICE_ID },
  });

  if (error || !data?.clientSecret) {
    throw new Error(data?.error ?? error?.message ?? 'Failed to create checkout session');
  }

  // 2. Initialize PaymentSheet
  const { error: initError } = await initPaymentSheet({
    paymentIntentClientSecret: data.clientSecret,
    customerEphemeralKeySecret: data.ephemeralKey,
    customerId: data.customer,
    merchantDisplayName: 'Mise',
    allowsDelayedPaymentMethods: false,
  });

  if (initError) {
    throw new Error(initError.message);
  }

  // 3. Present PaymentSheet
  const { error: presentError } = await presentPaymentSheet();

  if (presentError) {
    if (presentError.code === 'Canceled') {
      return false; // User canceled -- not an error
    }
    throw new Error(presentError.message);
  }

  // 4. Poll for subscription activation (webhook typically fires within 2s)
  const store = useSubscriptionStore.getState();
  for (let i = 0; i < 5; i++) {
    await new Promise((resolve) => setTimeout(resolve, 2000));
    await store.refreshSubscription();
    if (useSubscriptionStore.getState().isPremium()) {
      return true;
    }
  }

  // Payment succeeded but webhook hasn't arrived yet
  return true;
}

/**
 * Opens Stripe-hosted billing portal for subscription management.
 */
export async function openBillingPortal(): Promise<void> {
  const { data, error } = await supabase.functions.invoke('create-billing-portal');

  if (error || !data?.url) {
    throw new Error(data?.error ?? error?.message ?? 'Failed to open billing portal');
  }

  await Linking.openURL(data.url);
}
```
