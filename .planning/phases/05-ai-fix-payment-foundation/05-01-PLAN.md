```sql
CREATE TABLE ai_usage_log (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE NOT NULL,
  function_name TEXT NOT NULL,
  input_hash TEXT,
  cached_response JSONB,
  tokens_used INTEGER,
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE INDEX idx_ai_usage_user_date ON ai_usage_log (user_id, created_at);
CREATE INDEX idx_ai_usage_hash ON ai_usage_log (input_hash) WHERE input_hash IS NOT NULL;

ALTER TABLE ai_usage_log ENABLE ROW LEVEL SECURITY;
CREATE POLICY "ai_usage_select_own" ON ai_usage_log FOR SELECT USING (auth.uid() = user_id);

export class RateLimitError extends Error {
  limit: number;
  remaining: number;

  constructor(message: string, limit: number, remaining: number) {
    super(message);
    this.name = 'RateLimitError';
    this.limit = limit;
    this.remaining = remaining;
  }
}
```

2. **In parseRecipeUrl**, after the `supabase.functions.invoke` call, before the existing error handling:
```typescript
// Check for rate limit response
if (error) {
  // Supabase functions.invoke returns the response body in error.context for non-2xx
  const errorBody = typeof error.message === 'string' ? error.message : '';
  if (errorBody.includes('Rate limit exceeded') || error.context?.status === 429) {
    try {
      const parsed = JSON.parse(errorBody);
      throw new RateLimitError(
        `Daily import limit reached (${parsed.limit}/${parsed.limit})`,
        parsed.limit ?? 10,
        0
      );
    } catch (e) {
      if (e instanceof RateLimitError) throw e;
      throw new RateLimitError('Daily import limit reached. Try again tomorrow.', 10, 0);
    }
  }
}
```
