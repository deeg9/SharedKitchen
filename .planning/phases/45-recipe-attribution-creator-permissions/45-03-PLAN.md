```typescript
searchPeople: async (query: string) => {
  if (query.trim().length < 2) {
    set({ peopleResults: [], peopleLoading: false });
    return;
  }
  set({ peopleLoading: true });
  try {
    const { data, error } = await supabase
      .from('profiles')
      .select('id, handle, display_name, avatar_url')
      .or(`handle.ilike.%${query}%,display_name.ilike.%${query}%`)
      .not('handle', 'is', null)
      .limit(20);
    if (error) throw error;
    set({ peopleResults: data ?? [], peopleLoading: false });
  } catch (error) {
    set({ peopleLoading: false, error: getErrorMessage(error) });
  }
},
```

**Part C -- Create PeopleSearchTab component (src/components/social/PeopleSearchTab.tsx):**

```typescript
import React from 'react';
import { View, Text, FlatList, ActivityIndicator, StyleSheet } from 'react-native';
import { useRouter } from 'expo-router';
import { Keyboard } from 'react-native';
import { useSocialStore } from '../../stores/socialStore';
import { UserSearchResult } from './UserSearchResult';
import { colors } from '../../theme/colors';
import { typography } from '../../theme/typography';
import { spacing } from '../../theme/spacing';

interface PeopleSearchTabProps {
  onNavigate: () => void;  // Called when navigating away (to clear search + close overlay)
}

export function PeopleSearchTab({ onNavigate }: PeopleSearchTabProps) {
  const router = useRouter();
  const { peopleResults, peopleLoading, clearPeopleSearch } = useSocialStore();

  const handleResultPress = (userId: string) => {
    Keyboard.dismiss();
    clearPeopleSearch();
    onNavigate();
    router.push(`/user/${userId}` as any);
  };

  if (peopleLoading) {
    return (
      <View style={styles.statusContainer}>
        <ActivityIndicator size="small" color={colors.primary} />
      </View>
    );
  }

  if (peopleResults.length === 0) {
    return null; // Empty state handled by parent based on query length
  }

  return (
    <FlatList
      data={peopleResults}
      keyExtractor={(item) => item.id}
      renderItem={({ item }) => (
        <UserSearchResult
          id={item.id}
          handle={item.handle}
          displayName={item.display_name}
          avatarUrl={item.avatar_url}
          onPress={handleResultPress}
        />
      )}
      keyboardShouldPersistTaps="handled"
      style={styles.list}
    />
  );
}
```

Fetch profile + recipes in useEffect:
```typescript
useEffect(() => {
  if (!id) return;
  const loadProfile = async () => {
    setLoading(true);
    setError(null);
    try {
      // Fetch profile
      const { data: profileData, error: profileError } = await supabase
        .from('profiles')
        .select('handle, display_name, avatar_url')
        .eq('id', id)
        .maybeSingle();
      if (profileError) throw profileError;
      if (!profileData) { setError('Profile not found'); setLoading(false); return; }
      setProfile(profileData);

      // Fetch their recipes (only from households the current user can see -- RLS enforced)
      const { data: recipeData, error: recipeError } = await supabase
        .from('recipes')
        .select('id, name, image_url')
        .eq('created_by', id)
        .order('created_at', { ascending: false })
        .limit(50);
      if (recipeError) throw recipeError;
      setRecipes(recipeData ?? []);
    } catch (err: any) {
      setError(err.message ?? 'Failed to load profile');
    } finally {
      setLoading(false);
    }
  };
  loadProfile();
}, [id]);
```
