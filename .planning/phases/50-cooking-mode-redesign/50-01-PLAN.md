     const lower = text.toLowerCase();

     if (/chop|dice|mince|slice|peel|wash|rinse|measure|prep/.test(lower)) return 'PREP';
     if (/grill|sear|char/.test(lower)) return 'GRILLING';

     if (/mix|combine|whisk/.test(lower)) return 'MIXING';
     return 'COOK';
   }
