2. Update the Supabase query (line 86). The kitchen_profiles table is linked to households via household_id. To get the handle of the household admin (first member), use a nested select through the relationship chain. Try this approach:
   ```typescript
   const { data, error } = await supabase
     .from('kitchen_profiles')
     .select('id, display_name, avatar_url, households!inner(household_members!inner(profiles!inner(handle)))')
     .in('id', missingIds);
   ```
   
   However, if Supabase nested selects through 3 levels are problematic, use a simpler approach -- make a separate query:
   ```typescript
   // First get kitchen profiles
   const { data: kitchens, error } = await supabase
     .from('kitchen_profiles')
     .select('id, display_name, avatar_url, household_id')
     .in('id', missingIds);
   
   if (error || !kitchens) return;
   
   // Then get first member handle for each household
   const householdIds = kitchens.map(k => k.household_id);
   const { data: members } = await supabase
     .from('household_members')
     .select('household_id, profiles!inner(handle)')
     .in('household_id', householdIds)
     .order('created_at', { ascending: true });
   
   // Build household -> handle lookup (first member = admin)
   const handleByHousehold = new Map<string, string | null>();
   if (members) {
     for (const m of members) {
       if (!handleByHousehold.has(m.household_id)) {
         handleByHousehold.set(m.household_id, (m.profiles as any)?.handle ?? null);
       }
     }
   }
   ```
